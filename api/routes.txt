# Graphite Meter HTTP route pin, shared by the Rust workspace and the browser client.
#
# Format (one route per non-comment, non-blank line):
#   <name> | <path> | <kind>
#
#   name = the key every implementation uses for the route
#   path = the exact path the server mounts and the client requests
#   kind = http | ws | wt, the transport that reaches the route. A wt route is
#          an HTTP/3 extended-CONNECT session; under authentication it carries a
#          bearer grant or a single-use token minted at /wt/session.
#
# Routes are never transmitted: preflight advertises origins and transports only,
# so each site carries its own table and this file keeps the tables identical.
# Every route test loads THIS file as its fixture:
#   Rust: rust/proto/tests/codes.rs (proto::route, the table the server mounts and the client requests)
#         rust/server/tests/app/gate.rs (the routes the server mounts)
#   TS:   client/src/protocol/pins.test.ts (protocol/routes.ts ROUTES)
#

download | /download | http
upload | /upload | http
uploadSession | /upload/session | http
uploadProgress | /upload/progress | http
wtSession | /wt/session | http
wsSession | /ws/session | http
ping | /ws/ping | ws
wtDownload | /wt/download | wt
wtUpload | /wt/upload | wt
wtPing | /wt/ping | wt
